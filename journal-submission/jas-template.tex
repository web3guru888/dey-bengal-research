%% ── Fonts ────────────────────────────────────────────────────────────────────

%% Main font (Latin + extended Latin for transliteration diacritics)
\setmainfont{Noto Serif}[
  BoldFont       = Noto Serif Bold,
  ItalicFont     = Noto Serif Italic,
  BoldItalicFont = Noto Serif Bold Italic
]

%% Bengali (U+0980–U+09FF)
\newfontfamily\bengalifont{Noto Serif Bengali}[
  Script   = Bengali,
  BoldFont = Noto Serif Bengali Bold
]

%% Devanagari (U+0900–U+097F incl. danda U+0964)
%% Only appears in paper as देव — using Noto Sans Devanagari UI (no serif available)
\newfontfamily\devanagarifont{Noto Sans Devanagari UI}[Script=Devanagari]

%% Arrows & symbols (→ U+2192 etc.) — DejaVu Serif has full arrows coverage
\newfontfamily\arrowfont{DejaVu Serif}

%% ── ucharclasses transitions ─────────────────────────────────────────────────

%% Bengali block → Noto Serif Bengali
\setTransitionsFor{Bengali}{\bengalifont}{\normalfont}

%% Devanagari sub-blocks (ucharclasses splits it; cover all five)
\setTransitionsFor{DevanagariPreMarks}{\devanagarifont}{\normalfont}
\setTransitionsFor{DevanagariMarks}{\devanagarifont}{\normalfont}
\setTransitionsFor{DevanagariPostMarks}{\devanagarifont}{\normalfont}
\setTransitionsFor{DevanagariDanDa}{\devanagarifont}{\normalfont}
\setTransitionsFor{DevanagariPostDanDa}{\devanagarifont}{\normalfont}

%% Arrows block (U+2190–U+21FF) → DejaVu Serif
\setTransitionsFor{Arrows}{\arrowfont}{\normalfont}
